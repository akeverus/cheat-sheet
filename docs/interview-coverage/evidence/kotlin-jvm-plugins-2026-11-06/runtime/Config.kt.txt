// Config.kt
class Config {
    companion object {
        @JvmStatic
        fun createDefault() = Config()

        @JvmField
        val VERSION = "1.0"
    }

    @JvmOverloads
    fun connect(
        host: String = "localhost",
        port: Int = 8080,
        ssl: Boolean = false
    ): String = "$host:$port:$ssl"
}
