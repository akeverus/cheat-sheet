fun main() {
data class Point(val x: Int, val y: Int)
val p1 = Point(1, 2)
val p2 = Point(1, 2)
println(p1 == p2)  // true — data class генерирует equals по полям
println(p1 === p2) // false — созданы два экземпляра

class Box(val value: Int)
println(Box(1) == Box(1)) // false — обычный класс не переопределил equals
}
