fun equalAsDouble(a: Double?, b: Double?): Boolean = a == b
fun equalAsAny(a: Any?, b: Any?): Boolean = a == b

fun main() {
    println(equalAsDouble(Double.NaN, Double.NaN)) // false
    println(equalAsAny(Double.NaN, Double.NaN))    // true
    println(equalAsDouble(-0.0, 0.0))             // true
    println(equalAsAny(-0.0, 0.0))                // false
    println(equalAsDouble(null, null))            // true
}
