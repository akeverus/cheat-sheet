data class User(val name: String, val age: Int)

class Coordinates(val x: Int, val y: Int) {
    operator fun component1() = x
    operator fun component2() = y
}

fun getUserData(): Pair<Long, String> = 42L to "alice@example.test"

fun main() {
    val (name, age) = User("Alice", 30)
    println("$name / $age") // Alice / 30

    val (x, y) = Coordinates(10, 20)
    println("$x / $y") // 10 / 20 — обычный класс с componentN

    for ((key, value) in mapOf("a" to 1, "b" to 2)) {
        println("$key -> $value")
    }
    listOf(User("Alice", 30), User("Bob", 25))
        .forEach { (userName, userAge) -> println("$userName is $userAge") }

    val (_, email) = getUserData()
    println(email) // alice@example.test
}
