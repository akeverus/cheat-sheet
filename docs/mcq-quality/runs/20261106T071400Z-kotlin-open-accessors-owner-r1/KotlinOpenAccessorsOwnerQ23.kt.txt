class Gauge {
    var writes: Int = 0
        private set
    var value: Int = 0
        set(input) {
            require(input >= 0)
            field = input
            writes++
        }
    val doubled: Int
        get() = value * 2
}
fun main() {
    val gauge = Gauge()
    check(gauge.writes == 0) // initializer не вызывает setter
    gauge.value = 4
    check(gauge.value == 4 && gauge.doubled == 8 && gauge.writes == 1)
}
